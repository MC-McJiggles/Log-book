<script>
  let rowCount = 0;

  function pad(n){ return String(n).padStart(2,'0'); }

  function todayStr(){
    const d = new Date();
    return `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;
  }

  function addRow(dateVal){
    rowCount++;
    const tbody = document.getElementById('logBody');
    const tr = document.createElement('tr');
    tr.id = `row-${rowCount}`;
    const rid = rowCount;

    tr.innerHTML = `
      <td><span class="row-num">${rid}</span></td>
      <td class="td-date"><input type="date" value="${dateVal||''}" oninput="recalc()"></td>
      <td class="divider-col"><input type="time" oninput="calcRow(${rid})"></td>
      <td><input type="time" oninput="calcRow(${rid})"></td>
      <td class="td-notes"><input type="text" placeholder="Terminal / Route…"></td>
      <td class="divider-col"><input type="time" oninput="calcRow(${rid})"></td>
      <td><input type="time" oninput="calcRow(${rid})"></td>
      <td class="td-num"><input type="number" min="0" max="120" placeholder="—" readonly style="color:#999;" id="lmin-${rid}"></td>
      <td class="divider-col td-num"><input type="number" step="0.01" min="0" placeholder="—" readonly id="reg-${rid}" style="font-weight:600;"></td>
      <td class="td-num"><input type="number" step="0.01" min="0" placeholder="—" readonly id="ot-${rid}" style="color:#c0392b;font-weight:600;"></td>
      <td class="td-num"><input type="number" step="0.01" min="0" placeholder="—" readonly id="tot-${rid}" style="color:#2c3e50;font-weight:700;"></td>
      <td class="divider-col td-num"><input type="number" min="0" placeholder="0" id="miles-${rid}" oninput="recalc()"></td>
      <td class="td-num"><input type="number" min="0" placeholder="0" id="loads-${rid}" oninput="recalc()"></td>
      <td class="td-notes"><input type="text" placeholder="Comments…"></td>
      <td class="divider-col no-print" style="width:40px;">
        <button class="btn btn-danger" onclick="delRow(${rid})">✕</button>
      </td>
    `;
    tbody.appendChild(tr);
    recalc();
  }

  function delRow(rid){
    const tr = document.getElementById(`row-${rid}`);
    if(tr) tr.remove();
    recalc();
  }

  function timeToMins(val){
    if(!val) return null;
    const [h,m] = val.split(':').map(Number);
    return h*60 + m;
  }

  function calcRow(rid){
    const cells = document.getElementById(`row-${rid}`)?.querySelectorAll('input[type=time]');
    if(!cells || cells.length < 4) return;
    const [shiftIn, shiftOut, lunchOut, lunchIn] = Array.from(cells).map(c=>c.value);

    const si = timeToMins(shiftIn);
    const so = timeToMins(shiftOut);
    const lo = timeToMins(lunchOut);
    const li = timeToMins(lunchIn);

    let lunchMins = 0;
    if(lo !== null && li !== null && li > lo) lunchMins = li - lo;

    const lminEl = document.getElementById(`lmin-${rid}`);
    lminEl.value = lunchMins > 0 ? lunchMins : '';
    lminEl.style.color = lunchMins > 0 ? '#555' : '#ccc';

    let totalHrs = null, regHrs = 0, otHrs = 0;
    if(si !== null && so !== null){
      let worked = so - si;
      if(worked < 0) worked += 1440; // crosses midnight
      worked -= lunchMins;
      if(worked < 0) worked = 0;
      totalHrs = worked / 60;
      regHrs = Math.min(totalHrs, 8);
      otHrs = Math.max(0, totalHrs - 8);
    }

    const setEl = (id, val) => {
      const el = document.getElementById(id);
      if(el) el.value = val !== null ? val.toFixed(2) : '';
    };
    setEl(`reg-${rid}`, regHrs || null);
    setEl(`ot-${rid}`, otHrs > 0 ? otHrs : null);
    setEl(`tot-${rid}`, totalHrs);

    recalc();
  }

  function recalc(){
    let sumReg=0, sumOt=0, sumTot=0, sumMiles=0, sumLoads=0;

    document.querySelectorAll('#logBody tr').forEach(tr=>{
      const rid = tr.id.replace('row-','');
      const reg = parseFloat(document.getElementById(`reg-${rid}`)?.value)||0;
      const ot  = parseFloat(document.getElementById(`ot-${rid}`)?.value)||0;
      const tot = parseFloat(document.getElementById(`tot-${rid}`)?.value)||0;
      const mi  = parseFloat(document.getElementById(`miles-${rid}`)?.value)||0;
      const ld  = parseInt(document.getElementById(`loads-${rid}`)?.value)||0;
      sumReg += reg; sumOt += ot; sumTot += tot;
      sumMiles += mi; sumLoads += ld;
    });

    document.getElementById('totRegHrs').textContent  = sumReg.toFixed(2);
    document.getElementById('totOtHrs').textContent   = sumOt.toFixed(2);
    document.getElementById('totTotalHrs').textContent= sumTot.toFixed(2);
    document.getElementById('totMiles').textContent   = sumMiles.toFixed(0);
    document.getElementById('totLoads').textContent   = sumLoads;

    document.getElementById('sumRegHrs').textContent  = sumReg.toFixed(2);
    document.getElementById('sumOtHrs').textContent   = sumOt.toFixed(2);
    document.getElementById('sumMiles').textContent   = sumMiles.toFixed(0);
    document.getElementById('sumLoads').textContent   = sumLoads;

    const rate = parseFloat(document.getElementById('hourlyRate').value)||0;
    const gross = rate ? (sumReg * rate + sumOt * rate * 1.5).toFixed(2) : null;
    document.getElementById('sumPay').textContent = gross ? `$${Number(gross).toLocaleString('en-US',{minimumFractionDigits:2})}` : '—';
  }

  function exportCSV(){
    const header = ['Date','Time In','Time Out','Location/Route','Lunch Out','Lunch In','Lunch Min',
                    'Reg Hrs','OT Hrs','Total Hrs','Miles','Loads','Notes'];
    const rows = [header];
    document.querySelectorAll('#logBody tr').forEach(tr=>{
      const inputs = tr.querySelectorAll('input');
      if(!inputs.length) return;
      const vals = Array.from(inputs).map(i=>i.value);
      rows.push(vals.slice(0,13));
    });
    const worker = document.getElementById('workerName').value || 'WorkLog';
    const csv = rows.map(r=>r.map(v=>`"${v}"`).join(',')).join('\n');
    const blob = new Blob([csv], {type:'text/csv'});
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href=url; a.download=`${worker.replace(/\s+/g,'_')}_timelog.csv`;
    a.click(); URL.revokeObjectURL(url);
    showToast('CSV downloaded!');
  }

  function showToast(msg){
    const t = document.getElementById('toast');
    t.textContent = msg;
    t.style.display = 'block';
    setTimeout(()=>t.style.display='none', 2500);
  }

  // Pre-fill 7 rows for current week Mon–Sun
  function prefillWeek(){
    const today = new Date();
    const day = today.getDay(); // 0=Sun
    const monday = new Date(today);
    monday.setDate(today.getDate() - (day===0 ? 6 : day-1));
    for(let i=0;i<7;i++){
      const d = new Date(monday);
      d.setDate(monday.getDate()+i);
      const ds = `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;
      addRow(ds);
    }
    // Set week ending to Sunday
    const sun = new Date(monday);
    sun.setDate(monday.getDate()+6);
    document.getElementById('weekEnding').value =
      `${sun.getFullYear()}-${pad(sun.getMonth()+1)}-${pad(sun.getDate())}`;
  }

  prefillWeek();
</script>
